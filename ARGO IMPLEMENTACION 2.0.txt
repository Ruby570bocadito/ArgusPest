import networkx as nx
from typing import List, Tuple

class AStarPlanner:
    def __init__(self, knowledge_tree: 'KnowledgeTree'):
        self.graph = knowledge_tree.graph  # Grafo NetworkX con nodos y aristas ExploitEdge

    def find_routes(self, start_host_id: str, goal_description: str, profile: str) -> List['Route']:
        """
        Encuentra las mejores rutas desde start_host_id hacia hosts que satisfagan el goal.
        goal_description puede ser 'domain_admin', 'flag:CTF{*}', 'host:10.20.5.18', etc.
        """
        target_hosts = self._resolve_goal(goal_description)
        routes = []
        for target in target_hosts:
            try:
                # A* con heurística compuesta
                path = nx.astar_path(
                    self.graph,
                    start_host_id,
                    target,
                    heuristic=self._heuristic(profile),
                    weight='cost'
                )
                routes.append(self._build_route(path))
            except nx.NetworkXNoPath:
                continue
        # Ordenar por score combinado
        return sorted(routes, key=lambda r: r.score, reverse=True)

    def _heuristic(self, profile: str):
        """Heurística que combina distancia, sigilo y valor."""
        def h(n1, n2):
            host1 = self.graph.nodes[n1]['data']
            host2 = self.graph.nodes[n2]['data']
            distance = abs(int(host1.ip.split('.')[-1]) - int(host2.ip.split('.')[-1]))
            stealth_bonus = 1.0 if profile == 'ghost' else 0.5 if profile == 'balanced' else 0.0
            return distance * (1 - stealth_bonus)
        return h

    def _build_route(self, path: List[str]) -> 'Route':
        """Convierte un camino de nodos en una Route con pasos detallados."""
        steps = []
        total_probability = 1.0
        total_stealth = 0.0
        for i in range(len(path) - 1):
            edge = self.graph.edges[path[i], path[i+1]]['data']
            steps.append(PlanStep(
                source=path[i],
                target=path[i+1],
                technique=edge.technique,
                probability=edge.probability,
                stealth_cost=edge.stealth_cost
            ))
            total_probability *= edge.probability
            total_stealth += edge.stealth_cost
        return Route(steps=steps, total_probability=total_probability,
                     avg_stealth=total_stealth / len(steps) if steps else 1.0)

from qdrant_client import QdrantClient
from sentence_transformers import SentenceTransformer
import numpy as np

class CaseBasedReasoner:
    def __init__(self, qdrant_path: str = "./qdrant_data"):
        self.client = QdrantClient(path=qdrant_path)
        self.model = SentenceTransformer('all-MiniLM-L6-v2')  # 80 MB, CPU
        self._init_collection()

    def _init_collection(self):
        """Crea la colección si no existe."""
        from qdrant_client.models import Distance, VectorParams
        try:
            self.client.get_collection("cases")
        except:
            self.client.create_collection(
                collection_name="cases",
                vectors_config=VectorParams(size=384, distance=Distance.COSINE)
            )

    def add_case(self, service_description: str, action: str, success: bool, context: dict):
        """Añade un caso a la base de conocimiento."""
        vector = self.model.encode(service_description).tolist()
        self.client.upsert(
            collection_name="cases",
            points=[{
                "id": str(uuid.uuid4()),
                "vector": vector,
                "payload": {
                    "description": service_description,
                    "action": action,
                    "success": success,
                    "context": context
                }
            }]
        )

    def query_similar(self, service_description: str, top_k: int = 5) -> List[dict]:
        """Busca los k casos más similares."""
        vector = self.model.encode(service_description).tolist()
        results = self.client.search(
            collection_name="cases",
            query_vector=vector,
            limit=top_k
        )
        return [
            {
                "action": r.payload["action"],
                "success": r.payload["success"],
                "description": r.payload["description"],
                "score": r.score
            }
            for r in results
        ]

from anytree import Node, RenderTree

class RulesEngine:
    def __init__(self):
        self.root = Node("root")

    def evaluate(self, service: dict, defense_level: str, profile: str) -> List[dict]:
        """
        Evalúa el árbol de reglas y devuelve acciones ordenadas por prioridad.

        defense_level: 'none', 'low', 'medium', 'high'
        profile: 'ghost', 'balanced', 'blitz'

            actions.append({"action": "default_creds", "params": {"creds": ["root:root", "admin:admin"]}, "priority": 0.5})
        return sorted(actions, key=lambda a: a['priority'], reverse=True)
3.4 Fusionador de decisiones (core/decision_fusion.py)
python
class DecisionFusion:
    def __init__(self, config: dict):
        self.thresholds = config.get('thresholds', {
            'ghost': 0.85,
            'balanced': 0.7,
            'blitz': 0.4
        })

    def fuse(self, plan_routes: list, cbr_actions: list, rules_actions: list, profile: str) -> dict:
        """
        Combina las recomendaciones y decide la acción final.
        Retorna {'action': ..., 'confidence': ..., 'needs_approval': bool}
        """
        # Convertir cada sugerencia a una acción estandarizada con peso
        candidates = {}
        for route in plan_routes[:1]:  # Mejor ruta del planificador
            first_step = route.steps[0]
            candidates[first_step.technique] = candidates.get(first_step.technique, 0) + 0.5

        for cbr in cbr_actions[:2]:  # Top 2 casos similares
            if cbr['success']:
                candidates[cbr['action']] = candidates.get(cbr['action'], 0) + 0.3 * cbr['score']

        for rule in rules_actions[:2]:  # Top 2 reglas
            candidates[rule['action']] = candidates.get(rule['action'], 0) + 0.2 * rule['priority']

        # Seleccionar la acción con mayor puntuación
        if not candidates:
            return {'action': None, 'confidence': 0, 'needs_approval': True}
        best_action = max(candidates, key=candidates.get)
        confidence = min(1.0, candidates[best_action])
        threshold = self.thresholds[profile]
        return {
            'action': best_action,
            'confidence': confidence,
            'needs_approval': confidence < threshold
        }

5. Agentes
5.1 Stager Go mínimo (agents/stager/main.go)
Flujo:

Recopila AgentInfo (hostname, OS, IP, procesos corrientes).

Se conecta al C2 con gRPC + Chameleon.

Llama a Register.

Recibe RegisterAck con intervalo de beacon.

Bucle: espera, envía AgentEvent de latido, recibe DirectorCommand.

Si recibe DeployArsenalCommand, descarga y ejecuta el binario del arsenal en memoria.

5.2 Cell Go completo (agents/cell/)
El Cell es el agente persistente. Tiene módulos activables según el comando recibido:

recon.go: escaneo TCP SYN vía gopacket, banner grabbing HTTP/SMTP/SSH.

exploit.go: ejecuta exploits locales o invoca Metasploit vía shell msfconsole -x.

persist.go: lógica adaptativa (COM Hijacking, WMI, Registry, cron, systemd).

keylogger.go: hooks de teclado en Windows (SetWindowsHookEx) y Linux (/dev/input).

flag_hunter.go: búsqueda regex en disco, proc, env vars.

defense_shield.go: monitoreo de integridad de archivos y reinicio de servicios.

from pymetasploit3.msfrpc import MsfRpcClient

class MSFAgent:
    def __init__(self, password: str, port: int = 55553):
        self.client = MsfRpcClient(password, port=port)

    def execute_module(self, module_type: str, module_name: str, options: dict) -> dict:
        """Ejecuta un módulo de Metasploit y retorna el resultado."""
        module = self.client.modules.use(module_type, module_name)
        for key, value in options.items():
            module[key] = value
        job = module.execute()
        return {"job_id": job['job_id'], "uuid": job['uuid']}

    def get_sessions(self) -> list:
        """Lista las sesiones activas de Meterpreter."""
        return list(self.client.sessions.list.keys())
